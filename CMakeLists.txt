cmake_minimum_required(VERSION 3.24)
project(swak VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Network OpenGLWidgets Concurrent)
find_package(LibArchive REQUIRED)
include(CTest)
include(GNUInstallDirs)
option(SWAK_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(SWAK_COVERAGE "Enable GCC coverage for development" OFF)
if(SWAK_SANITIZERS AND NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require GCC or Clang.")
endif()
if(SWAK_COVERAGE AND (NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU" OR SWAK_SANITIZERS))
    message(FATAL_ERROR "Coverage requires GCC and a separate build without sanitizers.")
endif()

add_library(swak_build INTERFACE)
target_compile_features(swak_build INTERFACE cxx_std_20)
set(version_command "${CMAKE_COMMAND}" "-DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
    "-DOUTPUT_FILE=${CMAKE_CURRENT_BINARY_DIR}/generated/swak/version.h" "-DBASE_VERSION=${PROJECT_VERSION}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/version.cmake")
execute_process(COMMAND ${version_command} COMMAND_ERROR_IS_FATAL ANY)
add_custom_target(swak_version COMMAND ${version_command}
    BYPRODUCTS "${CMAKE_CURRENT_BINARY_DIR}/generated/swak/version.h" VERBATIM)
add_dependencies(swak_build swak_version)
target_include_directories(swak_build INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(swak_build INTERFACE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic;-Wconversion;-Wshadow;-fstack-protector-strong>)
target_link_options(swak_build INTERFACE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wl,-z,relro,-z,now>)
target_compile_options(swak_build INTERFACE
    $<$<AND:$<CONFIG:Release,RelWithDebInfo>,$<CXX_COMPILER_ID:GNU,Clang>>:-include;${CMAKE_CURRENT_SOURCE_DIR}/src/core/hardening.h>)
if(SWAK_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(swak_build INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(swak_build INTERFACE -fsanitize=address,undefined)
endif()
if(SWAK_COVERAGE)
    target_compile_options(swak_build INTERFACE --coverage -O0 -g)
    target_link_options(swak_build INTERFACE --coverage)
endif()

add_library(swak_core STATIC src/core/lan_protocol.cpp src/core/track_motion.cpp src/core/settings_dialog.cpp src/core/server_browser.cpp src/core/cot.cpp src/core/cot_stream.cpp src/core/identity.cpp src/core/servers.cpp src/core/ui_icons.cpp src/core/ui_icons.h src/core/polygon.cpp src/core/polygon.h src/core/map_auth.cpp src/core/marker_collections.cpp src/core/marker_collections.h src/core/milstd2525_symbols.cpp src/core/marker_symbols.cpp src/core/marker_symbols.h src/core/map_grid.cpp src/core/map_grid.h src/core/object_editor.cpp src/core/object_editor.h src/core/user_objects.cpp src/core/user_objects.h src/core/map_tools.cpp src/core/map_tools.h src/core/map_selection.cpp src/core/map_selection.h src/core/vector_tiles.cpp src/core/vector_tiles.h src/core/data.cpp src/core/data.h src/core/event.cpp src/core/event.h src/core/files.cpp src/core/files.h src/core/interfaces.h src/core/single_line_label.h
    src/core/plugin_catalog.cpp src/core/plugin_catalog.h src/core/activity.cpp src/core/activity.h src/core/store.cpp src/core/store.h src/core/projection.cpp src/core/projection.h src/core/preferences.cpp src/core/preferences.h src/core/settings.cpp src/core/settings.h src/core/maps.cpp src/core/maps.h src/core/tiles.cpp src/core/tiles.h src/core/logging.cpp src/core/logging.h)
target_include_directories(swak_core PUBLIC src)
target_link_libraries(swak_core PUBLIC swak_build LibArchive::LibArchive Qt6::Widgets Qt6::Network Qt6::Concurrent)

function(swak_plugin name file)
    add_library(${name} STATIC ${file})
    target_compile_definitions(${name} PRIVATE QT_STATICPLUGIN)
    target_link_libraries(${name} PUBLIC swak_core ${ARGN})
endfunction()
swak_plugin(swak_kml src/plugins/kml.cpp LibArchive::LibArchive)
swak_plugin(swak_mvt src/plugins/mvt.cpp)
target_include_directories(swak_mvt SYSTEM PRIVATE third_party/protozero/include)

swak_plugin(swak_basemap src/plugins/basemap.cpp)
